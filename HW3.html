import numpy as np
import matplotlib.pyplot as plt

def solve_q1():
    """Simulates a lightly damped oscillator and compares with analytical solution."""

    # Parameters
    m, k, b = 1.0, 10.0, 0.1  # mass, spring constant, damping (lightly damped)
    x0, v0 = 1.0, 0.0         # initial conditions (position/velocity)
    dt, T = 0.01, 20.0
    N = int(T / dt)
    t = np.linspace(0, T, N + 1)

    # Analytical solution for underdamped system
    gamma = b / (2 * m)
    omega0 = np.sqrt(k / m)
    if gamma >= omega0:
        print("System not underdamped.")
        return
    omega = np.sqrt(omega0**2 - gamma**2)
    phi = np.arctan2(-v0 - gamma * x0, omega * x0)
    A = x0 / np.cos(phi)
    x_analytical = A * np.exp(-gamma * t) * np.cos(omega * t + phi)

    # Simulation with Kick-Drift-Kick (Velocity Verlet)
    x, v = np.zeros(N + 1), np.zeros(N + 1)
    x[0], v[0] = x0, v0

    def F(x, v): return -k * x - b * v

    for i in range(N):
        v_half = v[i] + F(x[i], v[i]) * dt / (2 * m)
        x[i+1] = x[i] + v_half * dt
        v[i+1] = v_half + F(x[i+1], v_half) * dt / (2 * m)

    # Plotting
    plt.plot(t, x, '--', label='Simulated')
    plt.plot(t, x_analytical, label='Analytical')
    plt.xlabel('Time (s)')
    plt.ylabel('Position (m)')
    plt.title('Lightly Damped Oscillator: Sim vs Analytical')
    plt.legend()
    plt.grid()
    plt.show()

import numpy as np
import matplotlib.pyplot as plt

def solve_q2():
    """Simulate damped oscillator and plot total mechanical energy."""

    # Parameters
    m, k, b = 1.0, 10.0, 0.1
    x0, v0 = 1.0, 0.0
    dt, T = 0.01, 20.0
    N = int(T / dt)
    t = np.linspace(0, T, N + 1)

    # Initialize arrays
    x, v = np.zeros(N + 1), np.zeros(N + 1)
    x[0], v[0] = x0, v0

    def F(x, v): return -k * x - b * v

    for i in range(N):
        v_half = v[i] + F(x[i], v[i]) * dt / (2 * m)
        x[i+1] = x[i] + v_half * dt
        v[i+1] = v_half + F(x[i+1], v_half) * dt / (2 * m)

    # Compute total mechanical energy
    E = 0.5 * m * v**2 + 0.5 * k * x**2

    # Plot
    plt.plot(t, E, label='Total Energy', color='purple')
    plt.xlabel('Time (s)')
    plt.ylabel('Energy (J)')
    plt.title('Total Energy of Damped Oscillator')
    plt.grid()
    plt.legend()
    plt.show()

import numpy as np
import matplotlib.pyplot as plt

def solve_q4():
    """Simulate nonlinear oscillator (F = -k x |x|) and plot frequency vs k/m."""

    m, x0, v0 = 1.0, 1.0, 0.0
    dt, T = 0.001, 50.0
    N = int(T / dt)
    t = np.linspace(0, T, N + 1)
    ratios = np.linspace(0.1, 10, 20)
    freqs = []

    for r in ratios:
        k = r * m
        x, v = np.zeros(N + 1), np.zeros(N + 1)
        x[0], v[0] = x0, v0

        def F(x): return -k * x * abs(x)

        for i in range(N):
            v_half = v[i] + F(x[i]) * dt / (2 * m)
            x[i+1] = x[i] + v_half * dt
            v[i+1] = v_half + F(x[i+1]) * dt / (2 * m)

        # Zero crossings with positive velocity
        crossings = [t[i] for i in range(1, N) if x[i-1] * x[i] < 0 and v[i] > 0]

        if len(crossings) >= 2:
            periods = np.diff(crossings)
            freqs.append(1 / np.mean(periods))
        else:
            freqs.append(np.nan)

    # Plot
    plt.plot(ratios, freqs, 'o-', color='orange', label='Measured Frequency')
    plt.xlabel('k/m (1/s²)')
    plt.ylabel('Frequency (Hz)')
    plt.title('Natural Frequency vs k/m (Nonlinear Oscillator)')
    plt.grid()
    plt.legend()
    plt.show()

solve_q4()
